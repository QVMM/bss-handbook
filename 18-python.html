03数据分析：pandas 入门

本手册附带了一份练习数据 assets/健康数据-练习.csv（由 SmartCare 练习库的健康记录表导出，共 20 条）。把它和下面的脚本放在同一个文件夹中运行：

health.py
import pandas as pd

df = pd.read_csv("健康数据-练习.csv", encoding="utf-8")      # 读取 CSV
print(df.head())                                               # 前 5 行
print(df["RecordType"].value_counts())                         # 各类型记录数

bp = df[df["RecordType"] == "血压"]                            # 筛选血压
summary = bp.groupby("CustomerId")["Value1"].agg(["count", "max", "min", "mean"]).round(1)
print(summary)                                                 # 每位客户的收缩压统计

high = bp[bp["Value1"] >= 140]                                 # 收缩压 ≥ 140 的记录
print("偏高记录：", len(high))
high.to_csv("血压偏高.csv", index=False, encoding="utf-8-sig") # 导出，Excel 打开不乱码

运行结果中应能看到：血压 10 条、心率 5 条、血糖 5 条；客户 1 的收缩压共 7 次，最高 145.0、最低 128.0、平均 135.9；收缩压 ≥ 140 的记录 4 条，并生成文件“血压偏高.csv”。这和你在第 05 章用 SQL、第 08 章用 Java 算出的结果是同一批数据——三种工具做同一件事，正好可以互相核对。

04接口测试：只用标准库

Python 自带的 urllib 和 json 就能调用第 07 章的接口，适合快速写接口测试脚本：

api_test.py
import json
import urllib.request
import urllib.error

BASE = "http://localhost:5000"

def call(method, path, body=None):
    data = json.dumps(body).encode("utf-8") if body is not None else None
    req = urllib.request.Request(BASE + path, data=data, method=method,
                                 headers={"Content-Type": "application/json"})
    try:
        with urllib.request.urlopen(req) as resp:
            return resp.status, json.loads(resp.read().decode("utf-8"))
    except urllib.error.HTTPError as e:          # 404、500 等状态码
        return e.code, json.loads(e.read().decode("utf-8"))

cases = [
    ("登录成功", "POST", "/api/customer/login", {"phone": "13800000001", "password": "abc12345"}, True),
    ("密码错误", "POST", "/api/customer/login", {"phone": "13800000001", "password": "wrong"}, False),
    ("账号禁用", "POST", "/api/customer/login", {"phone": "13800000005", "password": "abc12345"}, False),
    ("分类列表", "GET", "/api/categories", None, True),
]
passed = 0
for name, method, path, body, expect in cases:
    status, result = call(method, path, body)
    ok = result.get("success") == expect
    passed += ok
    print(f"{'通过' if ok else '失败'}  {name}  {status}  {result.get('message')}")
print(f"共 {len(cases)} 条，通过 {passed} 条")
